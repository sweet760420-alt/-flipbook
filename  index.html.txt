    <script>
        // 控制書本開闔
        function toggleBook() {
            const book = document.getElementById('bookTarget');
            book.classList.toggle('open');
        }

        // --- 物理速度控制核心變數 ---
        let currentSpeed = 0;       // 當前動畫步進速度
        let targetSpeed = 0;        // 目標速度 (由使用者輸入決定)
        const baseSpeed = 0.2;      // 基礎背景微動速度
        const friction = 0.93;      // 摩擦力阻尼 (越接近 1 模擬越滑順，越小則停得越快)
        
        const speedMeter = document.getElementById('speedMeter');
        const eggContainer = document.getElementById('eggContainer');

        // 監聽滾輪事件 (不論向上向下滾動，皆轉換為正向動能)
        window.addEventListener('wheel', (e) => {
            targetSpeed += Math.abs(e.deltaY) * 0.08;
        }, { passive: true });

        // 監聽觸控與滑鼠拖拽滑動事件
        let lastY = 0;
        window.addEventListener('mousedown', (e) => { lastY = e.clientY; });
        window.addEventListener('mousemove', (e) => {
            if (e.buttons === 1) { // 當滑鼠左鍵按住拖曳時
                const deltaY = Math.abs(e.clientY - lastY);
                targetSpeed += deltaY * 0.15;
                lastY = e.clientY;
            }
        });
        window.addEventListener('touchstart', (e) => { lastY = e.touches[0].clientY; });
        window.addEventListener('touchmove', (e) => {
            const deltaY = Math.abs(e.touches[0].clientY - lastY);
            targetSpeed += deltaY * 0.2;
            lastY = e.touches[0].clientY;
        }, { passive: true });


        // --- Canvas 渲染引擎 ---
        const canvas = document.getElementById('eggCanvas');
        const ctx = canvas.getContext('2d');
        
        canvas.width = 150;
        canvas.height = 150;
        
        let virtualFrame = 0; // 改用虛擬幀計數，其遞增速度受 currentSpeed 控制
        const totalFrames = 300; 

        function drawFlipBook() {
            // 1. 物理運動學物理計算 (阻尼流體過渡)
            currentSpeed = currentSpeed * friction + targetSpeed * (1 - friction);
            targetSpeed *= 0.85; // 目標速度自然衰減
            
            // 計算總合播放速度 (基礎微動 + 讀者賦予的動能)
            const activeSpeed = baseSpeed + currentSpeed;
            virtualFrame += activeSpeed;

            // 更新儀表板顯示
            if (currentSpeed > 0.5) {
                speedMeter.classList.add('active');
                speedMeter.innerText = `當前翻頁速度：${(activeSpeed * 2.5).toFixed(2)} Hz (極速幻化中)`;
                // 讓 Canvas 容器隨翻頁速度產生微幅的 3D 震顫震盪感
                eggContainer.style.transform = `scale(${1 + Math.min(0.12, currentSpeed * 0.005)}) rotate(${Math.sin(virtualFrame*0.1)*2}deg)`;
            } else {
                speedMeter.classList.remove('active');
                speedMeter.innerText = `當前翻頁速度：${(activeSpeed * 2.5).toFixed(2)} Hz (等待指尖撥動)`;
                eggContainer.style.transform = `scale(1) rotate(0deg)`;
            }

            // 2. 開始繪製 Canvas
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.fillStyle = '#0a0a0c';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            
            const centerX = canvas.width / 2;
